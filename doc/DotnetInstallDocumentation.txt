# Create CA root https://benjamin-abt.com/blog/2025/10/13/create-custom-root-certificate-authority/
# Step 1: Create the Directory Structure
mkdir ./SeedySoft_Root_CA && cd SeedySoft_Root_CA

# Step 2: Create the CA Configuration File
cat > SeedySoft_Root_CA.cnf << EOL
[ req ]
default_bits            = 4096
default_md              = sha256
distinguished_name      = req_dn
prompt                  = no
x509_extensions         = v3_ca

[ req_dn ]
C                       = ES                    # countryName
CN                      = SeedySoft Root CA     # commonName
L                       = Burgos                # localityName
O                       = SeedySoft             # organizationName
OU                      = IT Crowd              # organizationalUnitName
ST                      = Burgos                # stateOrProvinceName
emailAddress            = seedysoft@gmail.com

[ v3_ca ]
basicConstraints        = critical, CA:true
keyUsage                = critical, digitalSignature, cRLSign, keyCertSign
subjectKeyIdentifier    = hash
authorityKeyIdentifier	= keyid:always,issuer
EOL

# Verify config file is ok
nano SeedySoft_Root_CA.cnf

# Step 3: Generate the Key
openssl genrsa -aes256 -out SeedySoft_Root_CA.key 4096
# SAVE PASSWORD !!!!
chmod g+r SeedySoft_Root_CA.key

# Step 4: Create the Root CA Certificate
openssl req -x509 -new -nodes -utf8 -sha256 -days 18250 \
  -config SeedySoft_Root_CA.cnf \
  -keyout SeedySoft_Root_CA.key \
  -out SeedySoft_Root_CA.crt
