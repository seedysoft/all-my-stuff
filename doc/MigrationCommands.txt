# Install tools (PS>):
dotnet tool update --global dotnet-ef

# SET VARIABLES (PS>):
$PrevMigrationName="Debug"
$NextMigrationName="Debug"

# Add a migration:
dotnet ef migrations add $NextMigrationName -v -c DbCxt                     -o .\Migrations                -p .\src\Libs\Infrastructure\Seedysoft.Libs.Infrastructure.csproj
dotnet ef migrations add $NextMigrationName -v -c GasStationPricesDbContext -o .\Infrastructure\Migrations -p .\src\Libs\GasStationPrices\Seedysoft.Libs.GasStationPrices.csproj

Add-Migration $NextMigrationName -Verbose -Context DbCxt                     -OutputDir .\Migrations                -Project Libs\Seedysoft.Libs.Infrastructure   -StartupProject Libs\Seedysoft.Libs.Infrastructure
Add-Migration $NextMigrationName -Verbose -Context GasStationPricesDbContext -OutputDir .\Infrastructure\Migrations -Project Libs\Seedysoft.Libs.GasStationPrices -StartupProject Libs\Seedysoft.Libs.GasStationPrices
```

# Remove latest migration:
dotnet ef migrations remove -v -p .\src\Libs\Infrastructure\Seedysoft.Libs.Infrastructure.csproj
dotnet ef migrations remove -v -p .\src\Libs\GasStationPrices\Seedysoft.Libs.GasStationPrices.csproj

Remove-Migration -Verbose -Context DbCxt                     -Project Libs\Seedysoft.Libs.Infrastructure   -StartupProject Libs\Seedysoft.Libs.Infrastructure
Remove-Migration -Verbose -Context GasStationPricesDbContext -Project Libs\Seedysoft.Libs.GasStationPrices -StartupProject Libs\Seedysoft.Libs.GasStationPrices
```

# List all existent migrations:
dotnet ef migrations list --no-build -v -c DbCxt                     -p .\src\Libs\Infrastructure\Seedysoft.Libs.Infrastructure.csproj
dotnet ef migrations list --no-build -v -c GasStationPricesDbContext -p .\src\Libs\GasStationPrices\Seedysoft.Libs.GasStationPrices.csproj

Get-Migration -Verbose -Context DbCxt                     -Project Libs\Seedysoft.Libs.Infrastructure
Get-Migration -Verbose -Context GasStationPricesDbContext -Project GasStationPrices\Seedysoft.Libs.GasStationPrices
```

# Execute a previous migration:
dotnet ef database update $PrevMigrationName --no-build -v -c DbCxt                     -p .\src\Libs\Infrastructure\Seedysoft.Libs.Infrastructure.csproj
dotnet ef database update $PrevMigrationName --no-build -v -c GasStationPricesDbContext -p .\src\Libs\GasStationPrices\Seedysoft.Libs.GasStationPrices.csproj

Update-Database $PrevMigrationName -Verbose -Context DbCxt                     -Project Libs\Seedysoft.Libs.Infrastructure
Update-Database $PrevMigrationName -Verbose -Context GasStationPricesDbContext -Project GasStationPrices\Seedysoft.Libs.GasStationPrices
```

# Obtain script
dotnet ef migrations script $PrevMigrationName $NextMigrationName -v -c DbCxt                     -p .\src\Libs\Infrastructure\Seedysoft.Libs.Infrastructure.csproj
dotnet ef migrations script $PrevMigrationName $NextMigrationName -v -c GasStationPricesDbContext -p .\src\Libs\GasStationPrices\Seedysoft.Libs.GasStationPrices.csproj
```
